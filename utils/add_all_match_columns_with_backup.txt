"""
Enhanced Wine Matching System with Multi-Stage Lookup
Integrates distributor, producer, and supplier reference tables for better matching
"""

import pandas as pd
import time
import datetime
import re
from difflib import SequenceMatcher

def clean_text_for_matching(text):
    """Clean text for better matching"""
    if pd.isna(text):
        return ""
    
    text = str(text).upper()
    
    # Remove common business suffixes
    suffixes_to_remove = [
        'INC', 'LLC', 'CO', 'CORP', 'CORPORATION', 'LTD', 'LIMITED',
        'COMPANY', 'WINERY', 'VINEYARDS', 'VINEYARD', 'WINES', 'WINE',
        'DISTRIBUTORS', 'DISTRIBUTING', 'IMPORTS', 'IMPORTERS'
    ]
    
    for suffix in suffixes_to_remove:
        text = re.sub(rf'\b{suffix}\b', '', text)
    
    # Clean up extra spaces and punctuation
    text = re.sub(r'[^\w\s]', ' ', text)
    text = ' '.join(text.split())
    
    return text.strip()

def extract_producer_from_description(item_description):
    """
    Extract producer name from item description with improved logic
    """
    desc = item_description.upper().strip()
    
    # Handle common abbreviations first
    abbreviation_map = {
        'CH ': 'CHATEAU ',
        'DOM ': 'DOMAINE ',
        'S/BLC': 'SAUVIGNON BLANC',
        'P/GRIG': 'PINOT GRIGIO', 
        'P/GRIS': 'PINOT GRIS',
        'S/BLANC': 'SAUVIGNON BLANC',
        'P/NOIR': 'PINOT NOIR'
    }
    
    for abbrev, full_name in abbreviation_map.items():
        desc = desc.replace(abbrev, full_name)
    
    words = desc.split()
    
    # Special handling for Chateau wines
    if 'CHATEAU' in words:
        idx = words.index('CHATEAU')
        if idx + 2 < len(words):
            return ' '.join(words[idx:idx+3])
        else:
            return ' '.join(words[idx:idx+2])
    
    # Special handling for Domaine wines
    if 'DOMAINE' in words:
        idx = words.index('DOMAINE')
        if idx + 2 < len(words):
            return ' '.join(words[idx:idx+3])
        else:
            return ' '.join(words[idx:idx+2])
    
    # For other wines, take first 2-3 words
    if len(words) >= 3:
        return ' '.join(words[:3])
    elif len(words) >= 2:
        return ' '.join(words[:2])
    else:
        return words[0] if words else ""

def find_best_text_match(search_text, candidate_list, threshold=0.6, max_candidates=50):
    """
    Find the best matching text from a list of candidates (with performance limits)
    """
    search_clean = clean_text_for_matching(search_text)
    if not search_clean:
        return None, 0
    
    best_match = None
    best_score = 0
    
    # Limit the number of candidates we fuzzy match against for performance
    candidates_to_check = candidate_list[:max_candidates] if len(candidate_list) > max_candidates else candidate_list
    
    for candidate in candidates_to_check:
        candidate_clean = clean_text_for_matching(candidate)
        if not candidate_clean:
            continue
            
        # Quick exact word check first (faster than fuzzy matching)
        search_words = set(search_clean.split())
        candidate_words = set(candidate_clean.split())
        
        # If no words match, skip expensive fuzzy matching
        if not search_words.intersection(candidate_words):
            continue
            
        # Calculate similarity
        similarity = SequenceMatcher(None, search_clean, candidate_clean).ratio()
        
        # Bonus for exact word matches
        if search_words.intersection(candidate_words):
            word_match_bonus = len(search_words.intersection(candidate_words)) / len(search_words)
            similarity += word_match_bonus * 0.3
        
        if similarity > best_score and similarity >= threshold:
            best_score = similarity
            best_match = candidate
    
    return best_match, best_score

def enhanced_wine_matching(df_sales, wine_review_data, distributors_df, 
                          wine_producers_df, suppliers_df, resume_from_checkpoint=None):
    """
    Enhanced wine matching using all reference tables
    """
    print("Starting Enhanced Wine Matching with Multi-Stage Lookup...")
    print(f"Sales data: {len(df_sales):,} rows")
    print(f"Wine reviews: {len(wine_review_data):,} rows")
    print(f"Distributors: {len(distributors_df):,} brands")
    print(f"Wine producers: {len(wine_producers_df):,} producers")
    print(f"Suppliers: {len(suppliers_df):,} suppliers")
    
    # Resume from checkpoint if provided
    if resume_from_checkpoint and os.path.exists(resume_from_checkpoint):
        print(f"🔄 Resuming from checkpoint: {resume_from_checkpoint}")
        df_enriched = pd.read_pickle(resume_from_checkpoint)
        
        # Find where to resume
        wine_mask = df_enriched['ITEM TYPE'] == 'WINE'
        processed_wines = df_enriched[wine_mask & (df_enriched['MATCH_STATUS'] != 'PENDING')]
        start_idx = len(processed_wines)
        print(f"📍 Resuming from wine #{start_idx:,}")
    else:
        df_enriched = df_sales.copy()
        start_idx = 0
        
        # Initialize new columns
        print("Initializing enrichment columns...")
        
        # Wine Review Data columns
        wine_review_columns = wine_review_data.columns.tolist()
        for col in wine_review_columns:
            if col in df_enriched.columns:
                new_col_name = f"wine_review_{col}"
            else:
                new_col_name = col
            df_enriched[new_col_name] = ""
        
        # Matching metadata columns
        df_enriched['EXTRACTED_PRODUCER'] = ""
        df_enriched['MATCHED_DISTRIBUTOR'] = ""
        df_enriched['MATCHED_WINE_PRODUCER'] = ""
        df_enriched['MATCHED_SUPPLIER'] = ""
        df_enriched['PRODUCER_MATCH_SCORE'] = 0.0
        df_enriched['SUPPLIER_MATCH_SCORE'] = 0.0
        df_enriched['WINE_REVIEW_MATCH_SCORE'] = 0.0
        df_enriched['MATCH_STATUS'] = 'PENDING'
    
    # Create lookup dictionaries for faster matching
    print("Creating lookup indices...")
    
    # Distributor lookup
    distributor_brands = distributors_df['Brand_Name'].dropna().tolist()
    
    # Producer lookup  
    wine_producer_names = wine_producers_df['Trade Name'].dropna().tolist()
    
    # Supplier lookup
    supplier_names = suppliers_df['Trade Name'].dropna().tolist()
    
    # Wine review lookup - create searchable index
    wine_review_titles = wine_review_data['title'].dropna().tolist()
    
    # Cache for performance
    description_cache = {}
    matched_count = 0
    cache_hits = 0
    
    # Process wines
    wine_mask = df_enriched['ITEM TYPE'] == 'WINE'
    wine_items = df_enriched[wine_mask].iloc[start_idx:]
    total_wines = len(df_enriched[wine_mask])
    
    print(f"Processing {len(wine_items)} remaining wines (of {total_wines} total)...")
    start_time = time.time()
    
    for i, (idx, row) in enumerate(wine_items.iterrows()):
        current_position = start_idx + i
        item_desc = row['ITEM DESCRIPTION']
        supplier = row['SUPPLIER']
        
        # Skip if already processed
        if row['MATCH_STATUS'] != 'PENDING':
            continue
        
        # Check cache first
        cache_key = f"{item_desc}_{supplier}"
        if cache_key in description_cache:
            result = description_cache[cache_key]
            cache_hits += 1
        else:
            # Multi-stage matching process
            result = perform_multi_stage_match(
                item_desc, supplier, 
                distributor_brands, wine_producer_names, supplier_names, 
                wine_review_data, distributors_df, wine_producers_df, suppliers_df
            )
            description_cache[cache_key] = result
        
        # Apply results to dataframe
        if result:
            # Update all matched data
            for key, value in result.items():
                if key in df_enriched.columns:
                    df_enriched.at[idx, key] = value
            
            df_enriched.at[idx, 'MATCH_STATUS'] = 'MATCHED'
            matched_count += 1
        else:
            df_enriched.at[idx, 'MATCH_STATUS'] = 'NO_MATCH'
        
        # Save checkpoint every 1000 items
        if (current_position + 1) % 1000 == 0:
            checkpoint_filename = f'enhanced_wine_matching_checkpoint_{current_position+1}.pkl'
            df_enriched.to_pickle(checkpoint_filename)
            print(f"*** 💾 CHECKPOINT SAVED: {checkpoint_filename} ***")
        
        # Progress reporting every 500 items
        if (current_position + 1) % 500 == 0:
            elapsed_time = time.time() - start_time
            processed = current_position + 1
            
            if processed > start_idx:
                avg_time = elapsed_time / (processed - start_idx)
                remaining = total_wines - processed
                eta_seconds = remaining * avg_time
                eta = str(datetime.timedelta(seconds=int(eta_seconds)))
            else:
                eta = "Calculating..."
            
            cache_efficiency = (cache_hits / (processed - start_idx)) * 100 if processed > start_idx else 0
            match_rate = (matched_count / (processed - start_idx)) * 100 if processed > start_idx else 0
            
            print(f"Progress: {processed:,}/{total_wines:,} ({processed/total_wines*100:.1f}%)")
            print(f"  Matches: {matched_count:,} ({match_rate:.1f}%)")
            print(f"  Cache efficiency: {cache_efficiency:.1f}%")
            print(f"  ETA: {eta}")
    
    # Final save
    final_filename = f'enhanced_wine_matching_FINAL_{datetime.datetime.now().strftime("%Y%m%d_%H%M")}.pkl'
    df_enriched.to_pickle(final_filename)
    
    # Final statistics
    total_time = time.time() - start_time
    final_match_rate = (matched_count / len(wine_items)) * 100 if len(wine_items) > 0 else 0
    
    print("\n" + "="*60)
    print("ENHANCED MATCHING RESULTS:")
    print(f"Total wines processed: {len(wine_items):,}")
    print(f"Matches found: {matched_count:,} ({final_match_rate:.1f}%)")
    print(f"Cache efficiency: {(cache_hits/len(wine_items)*100):.1f}%")
    print(f"Total runtime: {str(datetime.timedelta(seconds=int(total_time)))}")
    print(f"🎉 FINAL RESULTS SAVED: {final_filename}")
    
    return df_enriched

def perform_multi_stage_match(item_desc, supplier, distributor_brands, wine_producer_names, 
                             supplier_names, wine_review_data, distributors_df, 
                             wine_producers_df, suppliers_df):
    """
    Perform multi-stage matching across all reference tables
    """
    result = {}
    
    # Stage 1: Extract producer from item description
    extracted_producer = extract_producer_from_description(item_desc)
    result['EXTRACTED_PRODUCER'] = extracted_producer
    
    # Stage 2: Match against distributor brands
    dist_match, dist_score = find_best_text_match(extracted_producer, distributor_brands, threshold=0.7)
    if dist_match:
        # Get distributor info
        distributor_info = distributors_df[distributors_df['Brand_Name'] == dist_match].iloc[0]
        result['MATCHED_DISTRIBUTOR'] = distributor_info['Distributor']
    
    # Stage 3: Match against wine producers
    producer_match, producer_score = find_best_text_match(extracted_producer, wine_producer_names, threshold=0.6)
    if producer_match:
        result['MATCHED_WINE_PRODUCER'] = producer_match
        result['PRODUCER_MATCH_SCORE'] = producer_score
    
    # Stage 4: Match supplier
    supplier_match, supplier_score = find_best_text_match(supplier, supplier_names, threshold=0.6)
    if supplier_match:
        result['MATCHED_SUPPLIER'] = supplier_match
        result['SUPPLIER_MATCH_SCORE'] = supplier_score
    
    # Stage 5: Fast wine review matching using substring search first
    if extracted_producer and len(extracted_producer) > 2:
        # First, filter wine reviews using fast string contains
        potential_wine_matches = wine_review_data[
            wine_review_data['title'].str.contains(extracted_producer, case=False, na=False, regex=False)
        ]
        
        if len(potential_wine_matches) > 0:
            # Only do expensive fuzzy matching on the filtered subset
            wine_titles = potential_wine_matches['title'].tolist()
            wine_match, wine_score = find_best_text_match(extracted_producer, wine_titles, threshold=0.5)
            
            if wine_match:
                # Get full wine review data
                wine_data = potential_wine_matches[potential_wine_matches['title'] == wine_match].iloc[0]
                result['WINE_REVIEW_MATCH_SCORE'] = wine_score
                
                # Add all wine review columns
                for col in wine_review_data.columns:
                    if col in ['Unnamed: 0']:  # Skip index columns
                        continue
                    result[f'wine_review_{col}' if col in ['country', 'description'] else col] = wine_data[col]
    
    # Only return a match if we have decent scores
    if not result:
        return None
        
    has_good_producer_match = result.get('PRODUCER_MATCH_SCORE', 0) >= 0.6
    has_good_wine_match = result.get('WINE_REVIEW_MATCH_SCORE', 0) >= 0.5
    has_good_supplier_match = result.get('SUPPLIER_MATCH_SCORE', 0) >= 0.6
    
    if has_good_producer_match or has_good_wine_match or has_good_supplier_match:
        return result
    else:
        return None  # Not good enough matches

# Usage example:
def run_enhanced_matching(df_clean_sales, wine_review_data, distributors_df, 
                         wine_producers_df, suppliers_df):
    """
    Wrapper function to run the enhanced matching system
    """
    return enhanced_wine_matching(
        df_sales=df_clean_sales,
        wine_review_data=wine_review_data,
        distributors_df=distributors_df,
        wine_producers_df=wine_producers_df,
        suppliers_df=suppliers_df
    )
